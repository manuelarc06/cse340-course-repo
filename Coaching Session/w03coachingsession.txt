export async function getCategoryById(id) {
    const sql = `
        SELECT
            category_id,
            name
        FROM categories
        WHERE category_id = $1;
    `;

}

I handle errors globally in the main server file using Express middleware. I include both a 404 handler for missing routes and a global error handler for server-side errors, which renders custom error pages depending on the status code.
app.use((req, res, next) => {
    const err = new Error('Page Not Found');
    err.status = 404;
    next(err);
});

// Global error handler
app.use((err, req, res, next) => {
    console.error('Error occurred:', err.message);
    console.error('Stack trace:', err.stack);

    const status = err.status || 500;
    const template = status === 404 ? '404' : '500';

    const context = {
        title: status === 404 ? 'Page Not Found' : 'Server Error',
        error: err.message,
        stack: err.stack
    };

    res.status(status).render(`errors/${template}`, context);
});

AI
This is a solid implementation of Express error handling—you’ve got both pieces in place: a 404 “catch-all” middleware and a centralized error handler, which is exactly how MVC apps are typically structured.
A few things are working well here:
* Your 404 middleware correctly forwards an error when no route matches.
* Your global handler cleanly separates status codes and selects the correct template (`404` vs `500`).
* You’re logging both the message and stack trace, which is useful for debugging during development.
